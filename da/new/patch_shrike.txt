+from __future__ import annotations
+from collections.abc import Sequence
+import math
+
+
+def check_dimensions(a: int, b: int, /) -> None:
+    if a != b:
+        raise ValueError(f'different vector dimensions {a} and {b}')
+
+
+def l2_distance(a: Sequence[float], b: Sequence[float], /) -> float:
+    check_dimensions(len(a), len(b))
+    return math.sqrt(sum((x - y) * (x - y) for x, y in zip(a, b)))
+
+
+def inner_product(a: Sequence[float], b: Sequence[float], /) -> float:
+    check_dimensions(len(a), len(b))
+    return sum(x * y for x, y in zip(a, b))
+
+
+def cosine_distance(a: Sequence[float], b: Sequence[float], /) -> float:
+    check_dimensions(len(a), len(b))
+    dot = 0.0
+    norma = 0.0
+    normb = 0.0
+    for x, y in zip(a, b):
+        dot += x * y
+        norma += x * x
+        normb += y * y
+    return cosine_from_parts(dot, norma, normb)
+
+
+def cosine_from_parts(dot: float, norma: float, normb: float, /) -> float:
+    denom = math.sqrt(norma * normb)
+    if denom == 0 or math.isnan(dot) or math.isnan(denom):
+        return math.nan
+    similarity = dot / denom
+    if math.isnan(similarity):
+        return math.nan
+    similarity = max(-1.0, min(1.0, similarity))
+    return 1.0 - similarity
+
+
+def l1_distance(a: Sequence[float], b: Sequence[float], /) -> float:
+    check_dimensions(len(a), len(b))
+    return sum(abs(x - y) for x, y in zip(a, b))
