+def _format_shortest(v: float, max_digits: int, round_trip: Callable[[float], float], /) -> str:
+    if v == 0:
+        return '-0' if math.copysign(1.0, v) < 0 else '0'
+

+    for p in range(1, max_digits + 1):
+        s = '%.*e' % (p - 1, v)
+        try:
+            if round_trip(float(s)) == v:
+                break
+        except OverflowError:
+            pass

+    sign = ''
+    if mantissa[0] == '-':
+        sign = '-'
+        mantissa = mantissa[1:]

+    x = int(exp_str)

+        if x < 0:
