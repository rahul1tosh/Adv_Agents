+def _round_binary(value: Fraction, mant_bits: int, emin: int, emax: int, /) -> float:
+    n, d = value.numerator, value.denominator
+    e = n.bit_length() - d.bit_length()
+    if (n << -e if e < 0 else n) < (d << e if e > 0 else d):
+        e -= 1
+    if e > emax:
+        return math.inf
+    exp = max(e, emin) - (mant_bits - 1)
+    scaled = Fraction(n << -exp, d) if exp < 0 else Fraction(n, d << exp)
+    m = round(scaled)
+    if m == 0:
+        return 0.0
+    if m >= 1 << mant_bits and max(e, emin) + 1 > emax:
+        return math.inf
+    return math.ldexp(m, exp)
