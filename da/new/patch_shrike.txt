+def _lineage(*pairs: tuple[str, str, str, str]):
+    return [
+        (ColumnQualifierTuple(src_col, src_tbl), ColumnQualifierTuple(tgt_col, tgt_tbl))
+        for src_col, src_tbl, tgt_col, tgt_tbl in pairs
+    ]

+        ("bigquery", "UNION ALL CORRESPONDING"),
