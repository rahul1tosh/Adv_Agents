+def _round_float32(num: int, den: int) -> tuple[int, int]:
+    e = max(num.bit_length() - den.bit_length() - 24, -149)
+    while True:
+        if e >= 0:
+            d = den << e
+            q, r = divmod(num, d)
+        else:
+            d = den
+            q, r = divmod(num << -e, d)
+        if q >= 1 << 24:
+            e += 1
+            continue
+        break
+    if 2 * r > d or (2 * r == d and q & 1):
+        q += 1
+        if q == 1 << 24:
+            q >>= 1
+            e += 1
+    return q, e
