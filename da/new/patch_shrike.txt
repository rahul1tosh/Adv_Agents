+    def max_inner_product(self, other: list[float] | ndarray | Vector, /) -> float:
+        a, b = self._distance_operands(other)
+        return -float32_sum(float32_array(map(operator.mul, a, b)))

+    def cosine_distance(self, other: list[float] | ndarray | Vector, /) -> float:
+        a, b = self._distance_operands(other)
+        similarity = float32_sum(float32_array(map(operator.mul, a, b)))
+        norma = float32_sum(float32_array(map(operator.mul, a, a)))
+        normb = float32_sum(float32_array(map(operator.mul, b, b)))
+
+        denominator = math.sqrt(norma * normb)
+        if denominator != 0:
+            similarity /= denominator
+        elif similarity != 0 and not math.isnan(similarity):
+            similarity = math.copysign(math.inf, similarity)
+        else:
+            similarity = math.nan
+

+            similarity = 1.0
+        elif similarity < -1:
+            similarity = -1.0
+        return 1.0 - similarity

+def sql_distance(op: str, a: Any, b: Any, cast: str) -> float:
+    fa = '%t' if isinstance(a, str) else '%b'
+    fb = '%t' if isinstance(b, str) else '%b'
+    row = conn.execute(f'SELECT {fa}::{cast} {op.replace('%', '%%')} {fb}::{cast}', (a, b), binary=True).fetchone()
+    assert row is not None
+    return float(row[0])

+def sql_error(op: str, a: Any, b: Any, cast: str) -> str:
+    with pytest.raises(psycopg.DatabaseError) as error:
+        sql_distance(op, a, b, cast)
+    message = error.value.diag.message_primary
+    assert message is not None
+    return message

+            Vector([1]).cosine_distance([])
+
+    def test_too_many_dimensions(self) -> None:

+    def test_nan(self) -> None:

+        with pytest.raises(ValueError, match='^NaN not allowed in vector$'):
+            Vector([1, 2]).max_inner_product([math.nan, 2])
+        with pytest.raises(ValueError, match='^NaN not allowed in vector$'):

+        with pytest.raises(ValueError, match='^infinite value not allowed in vector$'):

+    def test_float32_overflow_input(self) -> None:

+    def test_float32_overflow(self) -> None:
+        assert Vector([3e38]).l2_distance([-3e38]) == math.inf
+        assert Vector([3e38]).l1_distance([-3e38]) == math.inf
+        assert Vector([3e38, 3e38]).l1_distance([0, 0]) == math.inf
+        assert Vector([3e38, 3e38]).max_inner_product([1, 1]) == -math.inf
+        assert Vector([2e19]).max_inner_product([2e19]) == -math.inf
+        assert math.isnan(Vector([2e19]).cosine_distance([2e19]))
+        assert math.isnan(Vector([3e38, -3e38]).cosine_distance([3e38, 3e38]))
+
+    def test_float32_underflow(self) -> None:
+        assert Vector([1e-50]).to_list() == [0.0]
+        assert Vector([1e-20]).max_inner_product([1e-20]) == -9.99994610111476e-41
+        assert Vector([1e-20]).l2_distance([0]) == 9.999973050521066e-21
+
+        value = Vector([1e-30]).max_inner_product([1e-30])
+        assert value == 0.0 and math.copysign(1, value) == -1
+        assert Vector([1e-30]).l2_distance([0]) == 0.0
+        assert Vector([1e-45]).max_inner_product([1]) == -1.401298464324817e-45
+
+    def test_cosine_distance_zero_vector(self) -> None:
+        assert math.isnan(Vector([0, 0]).cosine_distance([1, 2]))
+        assert math.isnan(Vector([1, 2]).cosine_distance([0, 0]))
+        assert math.isnan(Vector([0, 0]).cosine_distance([0, 0]))
+        assert math.isnan(Vector([-0.0]).cosine_distance([0]))

+    def test_cosine_distance_norm_underflow(self) -> None:
+        assert Vector([1e-30]).cosine_distance([1e19]) == 0.0
+        assert Vector([-1e-30]).cosine_distance([1e19]) == 2.0
+        assert Vector([1e-30, 1]).cosine_distance([1e19, 0]) == 1.0

+    def test_cosine_distance_clamp(self) -> None:
+        assert Vector([-1.5885683298110962]).cosine_distance([-4.821664810180664]) == 0.0
+        assert Vector([1.5885683298110962]).cosine_distance([-4.821664810180664]) == 2.0

+    def test_signed_zero(self) -> None:
+        def is_negative_zero(value: float) -> bool:
+            return value == 0 and math.copysign(1, value) == -1
+
+        def is_positive_zero(value: float) -> bool:
+            return value == 0 and math.copysign(1, value) == 1
+
+        assert is_negative_zero(Vector([0]).max_inner_product([0]))
+        assert is_negative_zero(Vector([-0.0]).max_inner_product([0]))
+        assert is_negative_zero(Vector([1]).max_inner_product([-0.0]))
+        assert is_negative_zero(Vector([-1, 1]).max_inner_product([1, 1]))
+        assert is_positive_zero(Vector([-0.0]).l2_distance([0]))
+        assert is_positive_zero(Vector([-0.0]).l1_distance([0]))
+        assert is_positive_zero(Vector([1]).cosine_distance([1]))
+        assert is_positive_zero(Vector([-0.0, 1]).cosine_distance([0, 1]))
+
+    def test_symmetric(self) -> None:
+        a = Vector([0.1, -0.2, 0.3])
+        b = Vector([-0.4, 0.5, 0.6])
+        assert a.l2_distance(b) == b.l2_distance(a)
+        assert a.max_inner_product(b) == b.max_inner_product(a)
+        assert a.cosine_distance(b) == b.cosine_distance(a)
+        assert a.l1_distance(b) == b.l1_distance(a)
