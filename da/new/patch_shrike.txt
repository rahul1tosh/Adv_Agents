+from __future__ import annotations
+from collections.abc import Iterable, Iterator, Sequence
+import math
+
+Pairs = Iterable[tuple[float, float]]
+
+
+def check_dimensions(left: int, right: int, /) -> None:
+    if left != right:
+        raise ValueError(f'different vector dimensions {left} and {right}')
+
+
+def dense_pairs(left: Sequence[float], right: Sequence[float], /) -> list[tuple[float, float]]:
+    check_dimensions(len(left), len(right))
+    return list(zip(left, right))
+
+
+def sparse_pairs(
+    left_dim: int,
+    left_indices: Sequence[int],
+    left_values: Sequence[float],
+    right_dim: int,
+    right_indices: Sequence[int],
+    right_values: Sequence[float],
+    /
+) -> list[tuple[float, float]]:
+    check_dimensions(left_dim, right_dim)
+    return list(_merge_sparse(left_indices, left_values, right_indices, right_values))
+
+
+def _merge_sparse(
+    left_indices: Sequence[int],
+    left_values: Sequence[float],
+    right_indices: Sequence[int],
+    right_values: Sequence[float],
+    /
+) -> Iterator[tuple[float, float]]:
+    i = 0
+    j = 0
+    while i < len(left_indices) and j < len(right_indices):
+        if left_indices[i] == right_indices[j]:
+            yield (left_values[i], right_values[j])
+            i += 1
+            j += 1
+        elif left_indices[i] < right_indices[j]:
+            yield (left_values[i], 0.0)
+            i += 1
+        else:
+            yield (0.0, right_values[j])
+            j += 1
+    for k in range(i, len(left_indices)):
+        yield (left_values[k], 0.0)
+    for k in range(j, len(right_indices)):
+        yield (0.0, right_values[k])
+
+
+def l2_distance(pairs: Pairs, /) -> float:
+    return math.sqrt(sum((a - b) * (a - b) for a, b in pairs))
+
+
+def inner_product(pairs: Pairs, /) -> float:
+    return sum(a * b for a, b in pairs)
+
+
+def max_inner_product(pairs: Pairs, /) -> float:
+    return -inner_product(pairs)
+
+
+def cosine_distance(pairs: Pairs, /) -> float:
+    dot = 0.0
+    norm_left = 0.0
+    norm_right = 0.0
+    for a, b in pairs:
+        dot += a * b
+        norm_left += a * a
+        norm_right += b * b
+
+    denominator = math.sqrt(norm_left * norm_right)
+    if denominator == 0:
+        return math.nan
+
+    similarity = dot / denominator
+    similarity = max(-1.0, min(1.0, similarity))
+    return 1.0 - similarity
+
+
+def l1_distance(pairs: Pairs, /) -> float:
+    return sum(abs(a - b) for a, b in pairs)
+
+
+def check_bit_lengths(left: int, right: int, /) -> None:
+    if left != right:
+        raise ValueError(f'different bit lengths {left} and {right}')
+
+
+def _bits_to_int(data: bytes, length: int, /) -> int:
+    return int.from_bytes(data, byteorder='big') >> (8 * len(data) - length)
+
+
+def _popcount(value: int, /) -> int:
+    return bin(value).count('1')
+
+
+def hamming_distance(left_length: int, left_data: bytes, right_length: int, right_data: bytes, /) -> float:
+    check_bit_lengths(left_length, right_length)
+    left = _bits_to_int(left_data, left_length)
+    right = _bits_to_int(right_data, right_length)
+    return float(_popcount(left ^ right))
+
+
+def jaccard_distance(left_length: int, left_data: bytes, right_length: int, right_data: bytes, /) -> float:
+    check_bit_lengths(left_length, right_length)
+    left = _bits_to_int(left_data, left_length)
+    right = _bits_to_int(right_data, right_length)
+    ab = _popcount(left & right)
+    if ab == 0:
+        return 1.0
+    return 1.0 - ab / (_popcount(left) + _popcount(right) - ab)
