+from __future__ import annotations
+from collections.abc import Iterable, Iterator, Sequence
+import math

+def check_dimensions(left: int, right: int, /) -> None:
+    if left != right:
+        raise ValueError(f'different vector dimensions {left} and {right}')
+
+
+def check_bit_lengths(left: int, right: int, /) -> None:
+    if left != right:
+        raise ValueError(f'different bit lengths {left} and {right}')

+def _iter_sparse_pairs(
+    a_indices: Sequence[int],
+    a_values: Sequence[float],
+    b_indices: Sequence[int],
+    b_values: Sequence[float],
+) -> Iterator[tuple[float, float]]:
+    i = 0
+    j = 0
+    len_a = len(a_indices)
+    len_b = len(b_indices)
+    while i < len_a and j < len_b:
+        idx_a = a_indices[i]
+        idx_b = b_indices[j]
+        if idx_a == idx_b:
+            yield (a_values[i], b_values[j])
+            i += 1
+            j += 1
+        elif idx_a < idx_b:
+            yield (a_values[i], 0.0)
+            i += 1
+        else:
+            yield (0.0, b_values[j])
+            j += 1
+    while i < len_a:
+        yield (a_values[i], 0.0)
+        i += 1
+    while j < len_b:
+        yield (0.0, b_values[j])
+        j += 1
+
+
+def _l2_distance_pairs(pairs: Iterable[tuple[float, float]]) -> float:
+    distance = 0.0
+    for x, y in pairs:
+        diff = x - y
+        distance += diff * diff
+    return math.sqrt(distance)
+
+
+def _max_inner_product_pairs(pairs: Iterable[tuple[float, float]]) -> float:
+    dot = 0.0
+    for x, y in pairs:
+        dot += x * y
+    return 0.0 - dot
+
+
+def _cosine_distance_pairs(pairs: Iterable[tuple[float, float]]) -> float:
+    dot = 0.0
+    norm_a = 0.0
+    norm_b = 0.0
+    for x, y in pairs:
+        dot += x * y
+        norm_a += x * x
+        norm_b += y * y
+    if norm_a == 0.0 or norm_b == 0.0:
+        return math.nan
+    product = norm_a * norm_b
+    if product == 0.0 or math.isinf(product):
+        denom = math.sqrt(norm_a) * math.sqrt(norm_b)
+    else:
+        denom = math.sqrt(product)
+    if denom == 0.0:
+        return math.nan
+    similarity = dot / denom
+    if similarity > 1.0:
+        similarity = 1.0
+    elif similarity < -1.0:
+        similarity = -1.0
+    return 1.0 - similarity
+
+
+def _l1_distance_pairs(pairs: Iterable[tuple[float, float]]) -> float:
+    distance = 0.0
+    for x, y in pairs:
+        distance += abs(x - y)
+    return distance
+
+
+def l2_distance(a: Sequence[float], b: Sequence[float], /) -> float:
+    check_dimensions(len(a), len(b))
+    return _l2_distance_pairs(zip(a, b))
+
+
+def max_inner_product(a: Sequence[float], b: Sequence[float], /) -> float:
+    check_dimensions(len(a), len(b))
+    return _max_inner_product_pairs(zip(a, b))
+
+
+def cosine_distance(a: Sequence[float], b: Sequence[float], /) -> float:
+    check_dimensions(len(a), len(b))
+    return _cosine_distance_pairs(zip(a, b))
+
+
+def l1_distance(a: Sequence[float], b: Sequence[float], /) -> float:
+    check_dimensions(len(a), len(b))
+    return _l1_distance_pairs(zip(a, b))
+
+
+def sparse_l2_distance(

+
+def sparse_max_inner_product(

+    b_dim: int,
+    b_indices: Sequence[int],
+    b_values: Sequence[float],

+) -> float:
+    check_dimensions(a_dim, b_dim)
+    return _max_inner_product_pairs(_iter_sparse_pairs(a_indices, a_values, b_indices, b_values))
+
+
+def sparse_cosine_distance(
+    a_dim: int,
+    a_indices: Sequence[int],
+    a_values: Sequence[float],
+    b_dim: int,
+    b_indices: Sequence[int],
+    b_values: Sequence[float],
+    /,
+) -> float:
+    check_dimensions(a_dim, b_dim)
+    return _cosine_distance_pairs(_iter_sparse_pairs(a_indices, a_values, b_indices, b_values))
+
+
+def sparse_l1_distance(
+    a_dim: int,
+    a_indices: Sequence[int],
+    a_values: Sequence[float],
+    b_dim: int,
+    b_indices: Sequence[int],
+    b_values: Sequence[float],
+    /,
+) -> float:
+    check_dimensions(a_dim, b_dim)
+    return _l1_distance_pairs(_iter_sparse_pairs(a_indices, a_values, b_indices, b_values))
+
+
+def _bit_ints(a_length: int, a_data: bytes, b_length: int, b_data: bytes) -> tuple[int, int]:
+    check_bit_lengths(a_length, b_length)
+    a_int = int.from_bytes(a_data, byteorder='big')
+    b_int = int.from_bytes(b_data, byteorder='big')
+    rem = a_length % 8
+    if rem != 0:
+        shift = 8 - rem
+        a_int >>= shift
+        b_int >>= shift
+    return a_int, b_int
+
+
+def hamming_distance(a_length: int, a_data: bytes, b_length: int, b_data: bytes, /) -> float:
+    a_int, b_int = _bit_ints(a_length, a_data, b_length, b_data)
+    return float((a_int ^ b_int).bit_count())
+
+
+def jaccard_distance(a_length: int, a_data: bytes, b_length: int, b_data: bytes, /) -> float:
+    a_int, b_int = _bit_ints(a_length, a_data, b_length, b_data)
+    ab = (a_int & b_int).bit_count()
+    if ab == 0:
+        return 1.0
+    union = (a_int | b_int).bit_count()
+    return 1.0 - (ab / union)
