+    def cosine_distance(self, other: list[float] | ndarray | Vector, /) -> float:
+        a, b = self._prepare_distance(other)
+        similarity = 0.0
+        norma = 0.0
+        normb = 0.0
+        for x, y in zip(a, b):
+            similarity = to_float32(similarity + to_float32(x * y))
+            norma = to_float32(norma + to_float32(x * x))
+            normb = to_float32(normb + to_float32(y * y))

+        if math.isnan(norm_product):
+            return math.nan

+        if denom == 0.0:
