+    def test_from_binary_validation(self) -> None:
+        with pytest.raises(ValueError, match='invalid length'):
+            Vector.from_binary(struct.pack('>hh', 16001, 0))
+
+        with pytest.raises(ValueError) as error:
+            Vector.from_binary(struct.pack('>hh', 16001, 1) + b'\x00' * (4 * 16001))
+        assert str(error.value) == 'expected unused to be 0'
+
+        nan_bytes = struct.pack('>f', float('nan'))
+        with pytest.raises(ValueError) as error:
+            Vector.from_binary(struct.pack('>hh', 16001, 0) + nan_bytes * 16001)
+        assert type(error.value) is ValueError
+        assert str(error.value) == 'vector cannot have more than 16000 dimensions'
+
+        assert Vector.from_binary(struct.pack('>hh', 16000, 0) + struct.pack('>f', 1.0) * 16000).dimensions() == 16000
+
+        snan_bytes = struct.pack('>I', 0x7f800001)
+        for data, msg in [
+            (struct.pack('>hh2f', 2, 0, float('nan'), float('inf')), 'NaN not allowed in vector'),
+            (struct.pack('>hh', 1, 0) + snan_bytes, 'NaN not allowed in vector'),
+            (struct.pack('>hh2f', 2, 0, float('inf'), float('nan')), 'infinite value not allowed in vector'),
+            (struct.pack('>hh2f', 2, 0, float('-inf'), float('nan')), 'infinite value not allowed in vector'),
+        ]:
+            with pytest.raises(ValueError) as error:
+                Vector.from_binary(data)
+            assert type(error.value) is ValueError
+            assert str(error.value) == msg
