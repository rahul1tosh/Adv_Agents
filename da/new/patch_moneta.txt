+def _format_decimal(s: str, /) -> str:
+    neg = s[0] == '-'
+    if neg:
+        s = s[1:]
+    mant, exp_s = s.split('e')
+    x = int(exp_s)
+    digits = mant.replace('.', '').rstrip('0')
+    if -4 <= x < 16:
+        k = len(digits)
+        if x < 0:
+            res = f'0.{"0" * (-x - 1)}{digits}'
+        elif x + 1 >= k:
+            res = f'{digits}{"0" * (x + 1 - k)}'
+        else:
+            res = f'{digits[:x + 1]}.{digits[x + 1:]}'
+    elif len(digits) > 1:
+        res = f'{digits[0]}.{digits[1:]}e{x:+03d}'
+    else:
+        res = f'{digits}e{x:+03d}'
+    return f'-{res}' if neg else res

+def _format_float(v: float, fmts: tuple[str, ...], packer: struct.Struct, /) -> str:
+    vb = packer.pack(v)
+    vf: float = packer.unpack(vb)[0]
+    if vf == 0.0:
+        return '-0' if math.copysign(1.0, vf) < 0 else '0'
+    for fmt in fmts:
+        s = fmt % vf
+        try:
+            if packer.pack(float(s)) == vb:
+                return _format_decimal(s)
+        except OverflowError:
+            pass
+    raise ValueError('expected finite float')
