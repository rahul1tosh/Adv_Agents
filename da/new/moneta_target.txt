+def assert_pg_roundtrip(obj: Vector | HalfVector | SparseVector | Bit, pgtype: str) -> None:
+    assert type(obj).from_text(obj.to_text()) == obj
+    assert type(obj).from_binary(obj.to_binary()) == obj
+
+    if isinstance(obj, Bit):
+        res_t = conn.execute(f'SELECT %t::{pgtype}', (obj,)).fetchone()
+        assert res_t is not None
+        assert Bit.from_text(res_t[0]) == obj
+        res_b = conn.execute(f'SELECT %b::{pgtype}', (obj,), binary=True).fetchone()
+        assert res_b is not None
+        assert Bit.from_binary(res_b[0]) == obj
+    else:
+        if obj.dimensions() == 0:
+            return
+        res_t = conn.execute(f'SELECT %t::{pgtype}', (obj,)).fetchone()
+        assert res_t is not None
+        assert res_t[0] == obj
+        res_b = conn.execute(f'SELECT %b::{pgtype}', (obj,), binary=True).fetchone()
+        assert res_b is not None
+        assert res_b[0] == obj
